<script>
let energy = 100;
let restCount = 0;

const durationMap = {30:10,60:20,120:35,180:50};
const intensityMap = {light:0.8,medium:1,heavy:1.3};

function calcCost(dur,intensity){
  return Math.round(durationMap[dur] * intensityMap[intensity]);
}

function calcRisk(cost,intensity){
  let base = {light:1,medium:2,heavy:3}[intensity];
  let risk = base + (cost/20) + ((100-energy)/25);

  if(risk <3) return "SAFE";
  if(risk <5) return "MODERATE";
  return "HIGH";
}

function previewTask(){
  let dur = document.getElementById('duration').value;
  let intensity = document.getElementById('intensity').value;

  let cost = calcCost(dur,intensity);
  let risk = calcRisk(cost,intensity);

  document.getElementById('preview').innerHTML =
    `⚡ Cost: ${cost}<br>⚠️ Risk: ${risk}`;
}

function addTask(){
  let name = document.getElementById('name').value;
  let dur = document.getElementById('duration').value;
  let intensity = document.getElementById('intensity').value;

  let cost = calcCost(dur,intensity);
  let risk = calcRisk(cost,intensity);

  let el = document.createElement('div');
  el.className = 'task';
  el.innerHTML = `
    <b>${name}</b><br>
    ⚡ ${cost} | ⚠️ ${risk}<br>
    <button onclick="doTask(${cost}, this)">Do</button>
  `;

  document.getElementById('tasks').appendChild(el);
}

function doTask(cost, btn){
  let failChance = 0;

  if(energy <30) failChance = 0.4;
  else if(energy <60) failChance = 0.2;
  else failChance = 0.1;

  let result = Math.random() < failChance;

  if(result){
    energy -= cost/2;
    alert('❌ Failed: You lost focus');
  } else {
    energy -= cost;
    alert('✅ Success');
  }

  if(energy <0) energy = 0;

  btn.parentElement.remove();
  updateEnergy();
}

function rest(type){
  let base = {short:10,nap:20,deep:30}[type];

  let modifier = 1 - (restCount * 0.2);
  if(modifier < 0.5) modifier = 0.5;

  let gain = Math.round(base * modifier);

  if(energy <30) gain *= 1.2;

  energy += gain;
  if(energy >100) energy = 100;

  restCount++;

  updateEnergy();
}

function updateEnergy(){
  document.getElementById('energyBar').style.width = energy + '%';
  document.getElementById('energyText').innerText = energy + ' / 100';
}
